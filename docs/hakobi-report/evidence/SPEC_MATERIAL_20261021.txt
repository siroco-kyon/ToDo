// 予定の進捗（今日時点で何%進んでいるはずか）と、実績との差（何日分の遅れか）の計算。React に依存しない。
// ガントの「遅れ」と報告タブの「進捗の遅れ」で同じ判定にするため、ガントの表示設定（遅れとみなす日数・土日祝を数えないか）も読む。

export interface ScheduleOptions {
  /** 予定より何日分以上遅れていたら「遅れ」にするか */
  behindThresholdDays: number
  /** true のとき土日・祝日を数えない */
  workingDaysOnly: boolean
  holidayNames: ReadonlyMap<string, string>
}

export interface ScheduleBar {
  startDate: string
  endDate: string
}

/** ガントの表示設定の保存先（GanttView.tsx と同じ） */
export const GANTT_VIEW_SETTINGS_STORAGE_KEY = 'gantt-view-settings'
export const BEHIND_THRESHOLD_OPTIONS = [1, 2, 3, 5, 7]
export const DEFAULT_BEHIND_THRESHOLD_DAYS = 2

function parseKey(dateKey: string): Date {
  const [year, month, day] = dateKey.slice(0, 10).split('-').map(Number)
  return new Date(year, month - 1, day)
}

function formatKey(date: Date): string {
  return `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, '0')}-${String(date.getDate()).padStart(2, '0')}`
}

function addDays(dateKey: string, days: number): string {
  const date = parseKey(dateKey)
  date.setDate(date.getDate() + days)
  return formatKey(date)
}

function diffCalendarDays(dateKey: string, baseKey: string): number {
  return Math.round((parseKey(dateKey).getTime() - parseKey(baseKey).getTime()) / 86400000)
}

function isCountedDay(dateKey: string, options: ScheduleOptions): boolean {
  if (!options.workingDaysOnly) return true
  const day = parseKey(dateKey).getDay()
  return day !== 0 && day !== 6 && !options.holidayNames.has(dateKey)
}

/** 開始日から終了日まで（両端を含む）の日数。workingDaysOnly のときは土日・祝日を除く */
export function countScheduleDays(startKey: string, endKey: string, options: ScheduleOptions): number {
  if (endKey < startKey) return 0
  if (!options.workingDaysOnly) return diffCalendarDays(endKey, startKey) + 1
  let count = 0
  for (let key = startKey; key <= endKey; key = addDays(key, 1)) {
    if (isCountedDay(key, options)) count += 1
  }
  return count
}

/**
 * 予定の進捗（今日時点で何%進んでいるはずか）と、期間の日数。
 * 期間のうち昨日までに過ぎた日数に、今日の半日分を足した割合を予定とする
 */
export function calculateSchedule(bar: ScheduleBar, todayKey: string, options: ScheduleOptions): { expectedProgress: number; totalDays: number } {
  let effective = options
  let totalDays = countScheduleDays(bar.startDate, bar.endDate, effective)
  // 期間がすべて休日のタスクは、暦日で数える
  if (totalDays === 0) {
    effective = { ...options, workingDaysOnly: false }
    totalDays = countScheduleDays(bar.startDate, bar.endDate, effective)
  }
  if (todayKey < bar.startDate) return { expectedProgress: 0, totalDays }
  if (todayKey > bar.endDate) return { expectedProgress: 100, totalDays }
  const passedDays = countScheduleDays(bar.startDate, addDays(todayKey, -1), effective) + (isCountedDay(todayKey, effective) ? 0.5 : 0)
  return { expectedProgress: Math.min(100, Math.max(0, Math.round((passedDays / totalDays) * 100))), totalDays }
}

/** 予定との差を期間の日数に換算した値（正なら遅れ。例: 10日間のタスクで 25% 足りなければ 2.5） */
export function behindDaysOf(progress: number, bar: ScheduleBar, todayKey: string, options: ScheduleOptions): { expectedProgress: number; behindDays: number } {
  const { expectedProgress, totalDays } = calculateSchedule(bar, todayKey, options)
  const actual = Math.min(100, Math.max(0, progress))
  return { expectedProgress, behindDays: ((expectedProgress - actual) / 100) * totalDays }
}

/** 予定の進捗を半分以上下回ったら、日数が短くても遅れとみなす（2日間のタスクの最終日に 0% など） */
export const BEHIND_PROGRESS_GAP = 50

/**
 * 遅れているか。予定との差が遅れとみなす日数以上か、予定の進捗を半分以上下回っているとき。
 * 期間の短いタスクは日数の差が小さく、日数だけでは最終日に 0% でも「順調」になってしまうため
 */
export function isBehindSchedule(expectedProgress: number, progress: number, behindDays: number, thresholdDays: number): boolean {
  if (behindDays <= 0) return false
  return behindDays >= thresholdDays || expectedProgress - progress >= BEHIND_PROGRESS_GAP
}

/**
 * 遅れを判定するタスクか（ガント・報告タブ・チームタブで共通）。
 * 完了・アーカイブ・保留中・ToDoリスト、開始日か期限のないタスクは判定しない
 */
export function isScheduleTracked(todo: { status: string; list_mode?: number | boolean | null; start_date: string | null; due_date: string | null }): boolean {
  if (todo.status === 'done' || todo.status === 'archived' || todo.status === 'on_hold' || todo.list_mode) return false
  return Boolean(todo.start_date && todo.due_date)
}

/** 遅れの日数の表記。1日に満たないときは「1日未満」と書き、丸めて大きく見せない */
export function behindDaysLabel(behindDays: number): string {
  return behindDays < 1 ? '1日未満の遅れ' : `約${Math.round(behindDays)}日遅れ`
}

/** ガントの表示設定から、遅れとみなす日数と土日祝を数えないかを読む */
export function loadScheduleSettings(): { behindThresholdDays: number; workingDaysOnly: boolean } {
  try {
    const parsed = JSON.parse(window.localStorage.getItem(GANTT_VIEW_SETTINGS_STORAGE_KEY) ?? 'null') as { behindThresholdDays?: unknown; workingDaysOnly?: unknown } | null
    return {
      behindThresholdDays: typeof parsed?.behindThresholdDays === 'number' && BEHIND_THRESHOLD_OPTIONS.includes(parsed.behindThresholdDays)
        ? parsed.behindThresholdDays
        : DEFAULT_BEHIND_THRESHOLD_DAYS,
      workingDaysOnly: typeof parsed?.workingDaysOnly === 'boolean' ? parsed.workingDaysOnly : false
    }
  } catch {
    return { behindThresholdDays: DEFAULT_BEHIND_THRESHOLD_DAYS, workingDaysOnly: false }
  }
}

  ).length

function toKey(year: number, month: number, day: number): string {
  return `${year}-${String(month).padStart(2, '0')}-${String(day).padStart(2, '0')}`
}

/** 日付（YYYY-MM-DD）を1周期ずらす。毎月・毎年はその月にない日を末日に丸める（2/29 の次の年は平年なら 2/28） */
export function nextRecurrenceDateKey(dateKey: string, recurrence: TodoRecurrence): string {
  const [year, month, day] = dateKey.slice(0, 10).split('-').map(Number)
  if (recurrence === 'daily' || recurrence === 'weekly') {
    const date = new Date(Date.UTC(year, month - 1, day + (recurrence === 'daily' ? 1 : 7)))
    return toKey(date.getUTCFullYear(), date.getUTCMonth() + 1, date.getUTCDate())
  }
  const targetYear = recurrence === 'yearly' ? year + 1 : year + Math.floor(month / 12)
  const targetMonth = recurrence === 'yearly' ? month : (month % 12) + 1
  const lastDay = new Date(Date.UTC(targetYear, targetMonth, 0)).getUTCDate()
  return toKey(targetYear, targetMonth, Math.min(day, lastDay))
}

/** 毎年の基準の月日（期限、なければ開始日）。例: 10/18 */
function yearlyMonthDay(todo: Pick<Todo, 'start_date' | 'due_date'>): string | null {
  const dateKey = (todo.due_date ?? todo.start_date)?.slice(0, 10)
  if (!dateKey || !/^\d{4}-\d{2}-\d{2}$/.test(dateKey)) return null
  return `${Number(dateKey.slice(5, 7))}/${Number(dateKey.slice(8, 10))}`
}

export function todoRecurrenceLabel(todo: Pick<Todo, 'recurrence' | 'start_date' | 'due_date'>): string | null {
  const label = recurrenceKindLabel(todo.recurrence)
  if (!label) return null
  if (todo.recurrence !== 'yearly') return label
  const monthDay = yearlyMonthDay(todo)
  return monthDay ? `${label} ${monthDay}` : label
}

/**

 * 土日をスキップする設定なら、土日に当たるときは月曜にずらす。祝日は画面では数えないので、
 * 祝日をスキップする設定のときは holidayMayShift を立てる（日付がさらに後ろへずれることがある）

export function nextRecurrenceDue(todo: RecurrenceSource): { kind: '期限' | '開始日'; dateKey: string; holidayMayShift: boolean } | null {
  const recurrence = TODO_RECURRENCE_OPTIONS.find((option) => option.value === todo.recurrence)?.value
  if (!recurrence) return null
  const base = todo.due_date ? { kind: '期限' as const, dateKey: todo.due_date } : todo.start_date ? { kind: '開始日' as const, dateKey: todo.start_date } : null
  if (!base) return null

  return { kind: base.kind, dateKey: next, holidayMayShift: Boolean(todo.recurrence_skip_holidays) }
}

/** 完了にした時刻（ISO）より後に作られた、同じ名前・同じ繰り返しの未完了のタスク（＝そのとき作られた次回分） */
export function findSpawnedNext(todos: Todo[], source: Pick<Todo, 'id' | 'title' | 'recurrence'>, completedAtIso: string): Todo | null {
  if (!recurrenceKindLabel(source.recurrence)) return null
  // 端末とサーバーの時計の差を見込んで、少し前から探す
  const since = new Date(new Date(completedAtIso).getTime() - 60_000).toISOString()
  return todos
    .filter((todo) => todo.id !== source.id && todo.title === source.title && todo.recurrence === source.recurrence
      && todo.status !== 'done' && todo.status !== 'archived' && todo.created_at >= since)
    .sort((a, b) => b.created_at.localeCompare(a.created_at))[0] ?? null
}

  completedAtIso: string,
  showToast: Toast,
  reload: () => Promise<unknown> | void
): Promise<void> {
  let spawned: Todo | null = null
  try {
    spawned = findSpawnedNext(await window.api.todoGetAll(), source, completedAtIso)
  } catch { /* 見つけられなくても取り消しの知らせは出す */ }
  if (!spawned) {
    showToast(`「${source.title}」の完了を取り消しました`)
    return
  }
  const next = spawned
  const when = next.start_date ? `開始 ${formatShortDate(next.start_date)}` : next.due_date ? `期限 ${formatShortDate(next.due_date)}` : '日付なし'
  showToast(`「${source.title}」の完了を取り消しました。作られた次回分（${when}）は残っています`, 'success', {
    label: '次回分をアーカイブ',
    run: () => {
      void window.api.todoArchive(next.id).then(

          await reload()
          showToast('次回分をアーカイブしました', 'success', {
            label: '元に戻す',
            run: () => { void window.api.todoUnarchive(next.id, next.status === 'archived' ? 'not_started' : next.status).then(() => reload()) }
          })
        },
        (error: unknown) => showToast(error instanceof Error ? error.message : '次回分をアーカイブできませんでした', 'error')
      )
    }
  })
}

# AGENTS.md

このファイルは、このリポジトリを扱うエージェント向けの簡易メモです。

## コマンド

```bash
# デスクトップ版（Electron）
npm install
npm run dev
npm run build
npm run dist

# チーム向けサーバー版（マルチユーザー Web）
npm --prefix server install   # サーバー依存（独立パッケージ）
npm run dev:server            # サーバー（ポート 4577）
npm run dev:web               # Web フロント（ポート 5273、/api と /ws を 4577 へプロキシ）
npm run build:web             # Web ビルド → dist-web/
npm run start:server          # dist-web/ を配信しつつサーバー起動
npm run publish:desktop       # npm run dist の後、dist/ のインストーラーをサーバーの配布フォルダへ公開
```

Web 版の型チェックは `npx tsc --noEmit -p web/tsconfig.json`。ルートに `typecheck` スクリプトは無い。

## アプリの方向性

- 現在はガントチャート中心のアプリ
- `カレンダー` と `概要` はメイン導線から外している
- 依存関係は Finish to Start を採用
- 依存関係はガント画面とタスク詳細画面の両方から編集できる
- データの動作モードは 2 つ。個人用（Electron・ローカル DB）と、チーム用（社内 LAN・サーバー DB）。チーム用はブラウザと HAKOBI デスクトップ版の両方から利用できる
- チーム接続の機能: 担当者割り当て、チームダッシュボード（今/期限）、管理者によるアカウント発行、リアルタイム同期。ブラウザとデスクトップ版で共通
- **デスクトップ版（`src/`）を壊さないこと。** 両モードは同じ React 画面を共有する

## 技術構成

- main process
  Electron + better-sqlite3
- renderer
  React + TypeScript（デスクトップ版・サーバー版で共通）
- renderer から DB を直接触らず、`window.api` 経由でアクセスする
  - 個人用デスクトップ版: Electron IPC（`src/preload/index.ts`）
  - チーム接続（ブラウザ・デスクトップ版）: HTTP/WebSocket（`web/lib/client.ts`）。デスクトップ版はサーバーで配信する Web 画面を読み込む
- server
  Express + ws + better-sqlite3（`tsx` で TypeScript を直接実行、ポート 4577）。`server/` は自前の `node_modules` を持つ独立パッケージ（Electron ABI と Node ABI の競合回避）

## 実装メモ

- タスク日付変更時は `db.ts` 側で依存関係の連鎖を再計算する
- 依存関係の追加、待機日数変更、削除でも後続タスクを再計算する
- ガントの表示設定は `localStorage` に保存している
- ガントは初回表示と「今日に戻る」の操作で今日へ移動する。編集・同期・期間変更・ズーム・表示幅変更では、現在見ている日付を保つ
- **`Api` 契約 = `typeof api`（`src/preload/index.ts`）。** メソッドを 1 つ足すと、`web/lib/client.ts` も実装しないとコンパイルが通らない（左右対称が型で強制される）。`src/main/ipc.ts` のハンドラも対で要る
- 型定義は二重管理: `src/main/db.ts`（Api 契約の元）と `src/renderer/src/types.ts`（renderer 用の独立コピー）。両者を同一構造に保つ
- 現在ユーザーの取得は `window.api.authGetCurrentUser()` を使う（個人用では `null`、チーム接続では認証済みユーザーを返す）。`web/auth` の `useCurrentUser()` は Web エントリ専用なので共通の `App.tsx` からは使わない
- マルチユーザー判定は `users.length > 0`、管理者判定は `currentUser?.role === 'admin'`。個人用ではどちらも false、デスクトップ版のチーム接続ではブラウザと同じ判定を使う
- 依存連鎖ロジックはサーバー版 `server/src/db/todos.ts` に同じものを移植している（両方を直す）
- チーム接続のデスクトップ版の OS 通知・未読バッジ・トレイのタスク一覧は、非表示のメイン画面の `DesktopCompanion` が判定して `window.desktop` へ渡す。main 側はサーバー API を呼ばない。新着かどうかは端末の localStorage（`hakobi-desktop-notified:<userId>`）で判定し、初回は既存の通知を出さない
- サーバーが配信する画面とインストール済みのデスクトップ版は別々に更新される。`DesktopBridge` に足したメソッドは任意（`notify?`）にして存在を確かめてから呼び、`DesktopState` に足した項目は古い画面が送らない前提で扱う
- デスクトップ版の配布（`/api/desktop-release`）はログイン不要。`server/src/index.ts` で `/api/health` の直後、`/api/auth`・`/api/users`・`dataRouter`（`/api` 以下はすべて認証必須）と SPA のフォールバックより前にマウントする。配布できない理由（配布フォルダのパスを含む）は管理者にだけ返す
- 配信するのは、検証済みの `latest.yml` に書かれた exe 1つだけ（ファイル名のパターン・サイズ・sha512 を確認）。リクエストのパスからファイル名を作らない。`latest.yml` やインストーラーが差し替わると、サーバーを再起動せずに読み直す
- 公開は `npm run publish:desktop`（`server/scripts/publish-desktop.ts`）だけで行う。インストーラーを先に置き、`latest.yml` を最後に差し替える。古いインストーラーは消さない。`electron-builder.yml` の `publish:` は `dist/latest.yml` を出力させるために残す（自動更新はしない）
- `DesktopContext.appVersion`（`src/shared/desktop.ts`）は任意。1.0.4 以前のアプリは送らないので、無いときは 1.0.4 以前として扱う（`LAST_VERSION_WITHOUT_APP_VERSION`）
- `/download` はログイン不要のページ。`web/main.tsx` で `AuthGate` より前に分岐して描画する
- 更新のお知らせ（`DesktopUpdateNotice`）はデスクトップ版のメイン画面（`isDesktopMain`）だけで出し、`App.tsx` には入れない。「後で」で閉じた版は localStorage の `hakobi-desktop-update-dismissed` に保存する
- タスク一覧の絞り込み（担当・カテゴリ・検索）はガント・カンバン・計画・概要・報告にも効く。効いているときは各画面の見出しの横に「絞り込み中」のチップ（`ListFilterChips`、`App.tsx` の `listFilters` を各画面に渡す）を出し、その場で解除できる。1行を取る帯にはしない。報告タブには一覧の担当絞り込みをかけず（`reportTodos`）、報告タブ自身の担当絞り込みで決める。進捗タブの投稿先の選択肢も絞り込み前のタスクから選ぶ
- タスク追加（`QuickAddModal`）の初期値は、担当＝一覧の担当絞り込みの人（絞っていなければ自分）、カテゴリ＝左で選んでいるもの、開始日＝今日。追加したタスクが今の絞り込みで見えないときは、`handleAdded` が「絞り込みを解除して開く」付きで知らせる
- 上部のタブは、表示中のタブをもう一度押しても画面を変えない（以前はタスク詳細へ戻るトグルだった）
- 進捗率のバーは `ProgressSlider` に揃えている（タスク詳細・サブタスク・報告タブ）。ドラッグは要素の pointer イベントで受け、ウィンドウ全体にリスナーを付けない
- ガントの保留ボタンは、保留中のタスクだけ紫の「▶」を常に出す。保留していない行の「Ⅱ」は `.tarbo-gantt-hold-toggle`（行にマウスを乗せたときだけ表示）
- 「集計から外す」カテゴリ（`is_private`）はアクセス制限ではない。画面では「プライベート」と書かず「集計外」と表記する。上部の「すべて表示／集計外を隠す」は集計外カテゴリがあるときだけ出す
- 用語は「期限」（締切と書かない）、「進捗ログ」（進捗メモと書かない）に揃える。説明＝目的、メモ＝今の状況（上書き）、進捗ログ＝日付つきで積み上がる報告
- 進捗タブの「新着」は、前回タブを見た時刻（`progressTimeline.lastSeenAt:<userId>`、`lib/progressActivity.ts`）より後にほかの人が書いた投稿・返信。タブを開いた時点と離れる時点で `serverNow()` を書き込む。上部タブの件数は `useProgressUnseenCount`（サーバー版のみ、`onDataChanged('progress')` で取り直す）
- 「@名前」の判定はサーバーの `server/src/mention.ts` と画面の `lib/progressActivity.ts`（`mentionsUser`・`MENTION_BEFORE`）で同じ規則にする（片方だけ直さない）。入力欄の候補は `MentionTextarea`
- 進捗ログ・返信のリアクションは `ReactionBar`（`LikeButton.tsx`）。👍 は常に出し、🎉 👀 🙏 は「☺＋」から。サーバー・デスクトップの DB は任意の絵文字を受け付ける

## 注意点

- `better-sqlite3` を使うので、依存インストール後にネイティブ再ビルドが必要
- 文字化けした日本語が残っていたら、README と UI 文言の両方を確認する
- ドキュメント更新時は、ガント中心の仕様と依存関係機能の説明を優先する
- UI 文言はすべて日本語
- サーバー版は社内 LAN 前提（インターネット公開は想定外、セキュリティは軽め）。アカウントは管理者が発行（自己登録なし）、パスワードは bcrypt、セッションは Cookie
- 個人用デスクトップ版（`src/`）を壊さない。共通の `App.tsx` などに手を入れるときは、チーム用機能をフラグ（`multiUser` / `isAdmin`）でガードする
